cmake_minimum_required(VERSION 3.14)
project(mini_kv_store CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(kv src/kvstore.cpp)
add_executable(kvcli src/main.cpp)
target_link_libraries(kvcli kv)
enable_testing()
add_executable(test_kvstore tests/test_kvstore.cpp)
target_link_libraries(test_kvstore kv)
add_test(NAME kvstore COMMAND test_kvstore)
